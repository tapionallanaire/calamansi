cmake_minimum_required(VERSION 3.25)
project(calamansi_nds LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
get_filename_component(UPSTREAM "${CMAKE_CURRENT_LIST_DIR}/../../../../externals/nintendo/melonds-android" ABSOLUTE)
set(CORE "${UPSTREAM}/melonDS-android-lib")
add_compile_definitions(MELONDS_GL_HEADER=<GLES3/gl32.h> EGL_EGLEXT_PROTOTYPES)
set(BUILD_QT_SDL OFF CACHE BOOL "" FORCE)
set(ENABLE_GDBSTUB OFF CACHE BOOL "" FORCE)
set(ENABLE_JIT ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_link_options("-Wl,-z,max-page-size=16384")
add_subdirectory("${CORE}" "${CMAKE_BINARY_DIR}/melonDS-android-lib")
# Patch only a configure-time copy; pinned gitlinks are never modified.
set(PATCH "${CMAKE_CURRENT_LIST_DIR}/patches/audio-open-error.patch")
file(SHA256 "${PATCH}" PATCH_HASH)
if(NOT PATCH_HASH STREQUAL "0b17af69fdb905fe02fce4c1888bd938d84ac686eb884e4785ab7a5edd335bec")
    message(FATAL_ERROR "NDS audio patch hash mismatch")
endif()
set(PATCH_ROOT "${CMAKE_BINARY_DIR}/patched-core")
file(MAKE_DIRECTORY "${PATCH_ROOT}/src/android")
configure_file("${CORE}/src/android/MelonDSAudio.cpp" "${PATCH_ROOT}/src/android/MelonDSAudio.cpp" COPYONLY)
find_program(GIT_EXECUTABLE git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" apply --no-index --whitespace=nowarn "${PATCH}"
    WORKING_DIRECTORY "${PATCH_ROOT}" RESULT_VARIABLE PATCH_RESULT ERROR_VARIABLE PATCH_ERROR)
if(NOT PATCH_RESULT EQUAL 0)
    message(FATAL_ERROR "NDS pinned audio patch did not apply: ${PATCH_ERROR}")
endif()
set(RENDERER_PATCH "${CMAKE_CURRENT_LIST_DIR}/patches/renderer-init-error.patch")
file(SHA256 "${RENDERER_PATCH}" RENDERER_PATCH_HASH)
if(NOT RENDERER_PATCH_HASH STREQUAL "752b6bbd486c8b136a401bc3be11675cfdb782af081f9ba4dd31b9966741db04")
    message(FATAL_ERROR "NDS renderer patch hash mismatch")
endif()
foreach(SOURCE MelonInstance.cpp OpenGLContext.cpp MelonDS.cpp)
    configure_file("${CORE}/src/android/${SOURCE}" "${PATCH_ROOT}/src/android/${SOURCE}" COPYONLY)
endforeach()
execute_process(COMMAND "${GIT_EXECUTABLE}" apply --no-index --whitespace=nowarn "${RENDERER_PATCH}"
    WORKING_DIRECTORY "${PATCH_ROOT}" RESULT_VARIABLE PATCH_RESULT ERROR_VARIABLE PATCH_ERROR)
if(NOT PATCH_RESULT EQUAL 0)
    message(FATAL_ERROR "NDS pinned renderer patch did not apply: ${PATCH_ERROR}")
endif()
get_target_property(CORE_ANDROID_SOURCES melonDS-lib SOURCES)
foreach(SOURCE MelonDSAudio.cpp MelonInstance.cpp OpenGLContext.cpp MelonDS.cpp)
    list(REMOVE_ITEM CORE_ANDROID_SOURCES "${SOURCE}" "${CORE}/src/android/${SOURCE}")
endforeach()
set_property(TARGET melonDS-lib PROPERTY SOURCES "${CORE_ANDROID_SOURCES}")
foreach(SOURCE MelonDSAudio.cpp MelonInstance.cpp OpenGLContext.cpp MelonDS.cpp)
    target_sources(melonDS-lib PRIVATE "${PATCH_ROOT}/src/android/${SOURCE}")
endforeach()
set(FRONTEND "${CMAKE_CURRENT_LIST_DIR}/src/main/cpp/frontend")
file(GLOB FRONTEND_SOURCES CONFIGURE_DEPENDS "${FRONTEND}/*.cpp" "${FRONTEND}/performancehint/*.cpp")
add_library(melonDS-android-frontend SHARED ${FRONTEND_SOURCES})
target_link_libraries(melonDS-android-frontend PRIVATE melonDS-lib)
